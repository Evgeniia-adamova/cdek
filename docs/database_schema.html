Представления (Views)

v_session_summary
sessions LEFT JOIN kpi_evaluations session_id, video_id, duration_sec, processed_at, pipeline_status, contract_number, overall_score, max_possible_score, score_percentage, traffic_light, overall_status, person_count, segment_count

combined_emotions JOIN persons session_id, person_label, interval_id, video_emotion, audio_sentiment, combined_emotion, combined_valence
v_kpi_detail
kpi_evaluation_items JOIN kpi_evaluations JOIN kpi_criteria JOIN kpi_categories session_id, category_name, criterion_name, score, max_score, status, details
